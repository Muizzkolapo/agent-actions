# --- Example Usage ---
if __name__ == "__main__":
    # Create dummy project structure and config files for testing
    current_dir = Path(".")
    project_base = current_dir / "temp_project_test"
    project_base.mkdir(exist_ok=True)

    # Mock templates directory (ConfigManager looks for os.getcwd()/templates by default)
    # We can override it or create it. Let's create it.
    templates_dir_main = current_dir / "templates"
    templates_dir_main.mkdir(exist_ok=True)
    
    # Create specific template dir for this test if ConfigManager's default changes
    # For now, assume ConfigManager uses current_dir / "templates"

    # Constructor Config (user_config) - e.g., my_pipeline.yaml
    constructor_file_path = project_base / "my_pipeline.yaml"
    user_pipeline_config = {
        "my_pipeline": [ # This key should match the filename stem "my_pipeline"
            {"name": "agent_A", "agent_type": "llm", "model": "model-A", "is_operational": True, "dependencies": []},
            {"name": "agent_B", "agent_type": "function", "code_path": "path/to/code_B.py", "is_operational": True, "dependencies": ["agent_A"]},
            {"name": "agent_C", "agent_type": "tool", "api_key": "key_C", "is_operational": False, "dependencies": ["agent_A"]}, # Inactive
            {"name": "agent_D", "agent_type": "llm", "model": "model-D", "is_operational": True, "dependencies": ["agent_B", "agent_C"]}, # Depends on inactive C
        ]
    }
    with open(constructor_file_path, 'w') as f:
        yaml.dump(user_pipeline_config, f)

    # Default Config
    default_config_file_path = project_base / "default_pipeline_config.yaml"
    default_pipeline_config_content = {
        "default_agent_config": {
            "version": "1.0",
            "author": "DefaultAuthor",
            "is_operational": True # Default operational status
        }
    }
    with open(default_config_file_path, 'w') as f:
        yaml.dump(default_pipeline_config_content, f)

    logger.info(f"--- Test 1: Successful Load and Validation (expecting operational dependency error) ---")
    try:
        manager = ConfigManager(
            constructor_path=str(constructor_file_path),
            default_path=str(default_config_file_path),
            template_base_dir=str(templates_dir_main) # Pass the templates dir
        )
        manager.process_and_validate_configs() # This will call load, validate name, merge, determine order (which includes dep validation)
        
        print(f"Successfully processed. Agent Name: {manager.agent_name}")
        print(f"User Config Loaded: {'Yes' if manager.user_config else 'No'}")
        print(f"Default Config Loaded: {'Yes' if manager.default_config else 'No'}")
        print(f"Merged Agent Configs (agent_B): {manager.agent_configs.get('agent_B')}")
        print(f"Execution Order: {manager.execution_order}")

    except ValueError as e:
        print(f"!!! Test 1 FAILED as expected or an unexpected ValueError: {e}")
    except Exception as e:
        print(f"!!! Test 1 FAILED UNEXPECTEDLY: {type(e).__name__} - {e}")
        logger.error("Test 1 unexpected error", exc_info=True)


    logger.info(f"\n--- Test 2: Modify user_config to fix operational dependency ---")
    user_pipeline_config_fixed = {
        "my_pipeline": [
            {"name": "agent_A", "agent_type": "llm", "model": "model-A", "is_operational": True, "dependencies": []},
            {"name": "agent_B", "agent_type": "function", "code_path": "path/to/code_B.py", "is_operational": True, "dependencies": ["agent_A"]},
            {"name": "agent_C", "agent_type": "tool", "api_key": "key_C", "is_operational": True, "dependencies": ["agent_A"]}, # Now active
            {"name": "agent_D", "agent_type": "llm", "model": "model-D", "is_operational": True, "dependencies": ["agent_B", "agent_C"]},
        ]
    }
    with open(constructor_file_path, 'w') as f:
        yaml.dump(user_pipeline_config_fixed, f)
    
    try:
        manager_fixed = ConfigManager(
            constructor_path=str(constructor_file_path),
            default_path=str(default_config_file_path),
            template_base_dir=str(templates_dir_main)
        )
        manager_fixed.process_and_validate_configs()
        print(f"Successfully processed (fixed). Agent Name: {manager_fixed.agent_name}")
        print(f"Execution Order (fixed): {manager_fixed.execution_order}")
        # Expected order might be [agent_A, agent_C, agent_B, agent_D] or [agent_A, agent_B, agent_C, agent_D] (A, B/C in parallel)
        assert "agent_A" in manager_fixed.execution_order
        assert "agent_B" in manager_fixed.execution_order
        assert "agent_C" in manager_fixed.execution_order
        assert "agent_D" in manager_fixed.execution_order
        print("Test 2 assertions passed for execution order content.")

    except ValueError as e:
        print(f"!!! Test 2 FAILED: {e}")
        logger.error("Test 2 ValueError", exc_info=True)
    except Exception as e:
        print(f"!!! Test 2 FAILED UNEXPECTEDLY: {type(e).__name__} - {e}")
        logger.error("Test 2 unexpected error", exc_info=True)


    # Clean up dummy files and directories
    import shutil
    if constructor_file_path.exists(): constructor_file_path.unlink()
    if default_config_file_path.exists(): default_config_file_path.unlink()
    if project_base.exists(): shutil.rmtree(project_base)
    if templates_dir_main.exists() and not list(templates_dir_main.iterdir()): # Remove if empty
         shutil.rmtree(templates_dir_main)
    elif templates_dir_main.exists():
        print(f"Note: Template directory {templates_dir_main} was not empty, not removed.")
